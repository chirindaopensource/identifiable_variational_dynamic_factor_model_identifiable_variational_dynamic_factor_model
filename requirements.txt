# ── Section 1: Scientific Computing Core ─────────────────────────────────────
#
# These three packages form the numerical substrate of the entire iVDFM stack.
# They must be installed together and their versions kept mutually compatible
# (numpy ABI stability is a hard constraint for both scipy and scikit-learn).
# =============================================================================

# numpy — N-dimensional array engine.
#   Used for: NDArray arithmetic, np.linalg (eigenvalues of companion matrix,
#   QR factorisation for diagonal-restriction analysis), np.fft (FFT-based
#   impulse-response convolution), random seed control, and all tensor ↔ numpy
#   bridge operations via torch.from_numpy / .numpy().
#   Minimum 2.0.0: numpy 2.0 introduced the numpy 2.x C-API (NEP 50 type
#   promotion rules), the new `numpy.dtypes` module, and removed a large set
#   of deprecated aliases (np.bool, np.int, np.float, np.complex, np.object,
#   np.str) that would cause silent errors in 1.x code migrated here.
numpy>=2.0.0,<3.0.0

# pandas — labelled time-series data management.
#   Used for: pd.DataFrame / pd.Series construction, time-indexed rolling
#   windows, CSV/Parquet I/O for benchmark datasets (ETTh1, Weather), and
#   descriptive statistics in ablation reporting.
#   Minimum 2.0.0: pandas 2.0 introduced the Copy-on-Write (CoW) memory model
#   and the ArrowDtype backend, both of which affect in-place mutation
#   semantics that the module's data-pipeline code depends on.
pandas>=2.0.0,<4.0.0

# scipy — scientific algorithms.
#   Used for:
#     scipy.stats          — Laplace / Normal / Student-t PDF/CDF evaluation
#                            for the innovation prior; KL-divergence closed-form
#                            computation under exponential-family parametrisation.
#     scipy.optimize.linear_sum_assignment
#                          — Kuhn-Munkres (Hungarian) algorithm for max-weight
#                            bipartite matching during MCC (Mean Correlation
#                            Coefficient) computation under permutation
#                            invariance.
#   Minimum 1.11.0: scipy 1.11 stabilised the `scipy.stats` hypothesis-testing
#   API overhaul (method= kwarg) and the `linear_sum_assignment` gain/loss
#   matrix sign convention used in permutation-aligned metric calculation.
scipy>=1.11.0,<2.0.0


# ── Section 2: Machine Learning Utilities ────────────────────────────────────
#
# scikit-learn provides classical ML decomposition methods used for
# cold-start initialisation of the iVDFM latent states, not as the primary
# modelling engine.
# =============================================================================

# scikit-learn — classical ML / decomposition.
#   Used for:
#     sklearn.decomposition.PCA          — single-window PCA warm-start for the
#                                          initial factor state f_0, stabilising
#                                          the early phases of variational
#                                          training (see cold-start protocol).
#     sklearn.decomposition.FactorAnalysis
#                                        — OLS-style initialisation for AR
#                                          coefficients in the companion-matrix
#                                          transition block; provides a
#                                          statistically grounded starting point
#                                          for the diagonal AR parameters.
#   Minimum 1.3.0: introduces the `set_output` transformer API and the
#   validated `n_components` interface used in the PCA/FA warm-start routines.
scikit-learn>=1.3.0,<2.0.0

# torch — automatic differentiation and neural-network library.
#   Used for:
#     torch.nn.Module / nn.Linear / nn.LayerNorm / nn.Dropout
#                                — iVDFM sub-module hierarchy.
#     torch.nn.functional        — F.softmax (RegimeNet simplex projection),
#                                  F.relu (ablation-validated activation),
#                                  F.mse_loss (reconstruction term in ELBO).
#     torch.optim.Adam           — primary optimiser with weight-decay for
#                                  the full ELBO training loop.
#     torch.utils.data.TensorDataset / DataLoader
#                                — mini-batch iteration over (y, u, e)
#                                  triplets; stride-based windowing for
#                                  rolling-origin benchmark evaluation.

#                                  eps sampling for the Laplace innovation prior.
#   Minimum 2.0.0: introduces torch.compile (torch.dynamo backend) used for
#   JIT compilation of the FFT impulse-response convolution loop; stable
#   support for torch.linalg (eigenvalue-constrained companion matrix checks);
#   and the revised TensorDataset type-checking that the DataLoader relies on.
torch>=2.0.0,<3.0.0
